{{- /* Images from the post's own folder resolve to their published path; raster ones get width/height to avoid layout shift. A standalone image with a title ("![alt](img.png "Caption")") becomes a figure with that caption. */ -}} {{- $src := .Destination -}} {{- $size := "" -}} {{- with .PageInner.Resources.Get .Destination -}} {{- $src = .RelPermalink -}} {{- if in (slice "jpeg" "png" "webp" "gif") .MediaType.SubType -}} {{- $size = printf " width=%q height=%q" (string .Width) (string .Height) -}} {{- end -}} {{- end -}} {{- $img := printf "" $src .PlainText $size -}} {{- if and .IsBlock .Title -}} {{- else -}} {{ $img | safeHTML }} {{- end -}}
